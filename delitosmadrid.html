# Función de normalización para evitar los NA 
limpiar_todo <- function(texto) {
  texto <- toupper(texto)
  texto <- iconv(texto, to="ASCII//TRANSLIT") 
  texto <- str_replace_all(texto, "[^[:alnum:]]", "") 
  texto <- trimws(texto)
  return(texto)
}

  read_excel(paste0(mes, ".xlsx"), sheet = "SEGURIDAD", skip = 2) %>%
    select(DISTRITO = 1, PERS = 2, PATRI = 3) %>%
